function tsunamiColor(level){ return (TSU_LEVELS[level] ? TSU_LEVELS[level].color : "rgba(255,255,255,.22)"); } function fmtM(x){ return (x>=10 ? x.toFixed(0) : x.toFixed(1)); } function isSeaAt(lat, lon){ if (!PrefDB.ready) return false; return (findPrefId(lat, lon) < 0); } function tsunamiScore(M, depthKm){ const shallow = clamp(1 - depthKm/120, 0, 1); let s = (M - 6.0) - 0.006*depthKm + 0.35*shallow; if (depthKm >= 100) s -= 0.9; // deep suppress return s; } function tsunamiLevelFrom(M, depthKm){ // baseline from score, then enforce minimum for M>=7 shallow-ish seabed const s = tsunamiScore(M, depthKm); let lvl = 0; if (s < 0.90) lvl = 0; else if (s < 1.60) lvl = 1; else if (s < 2.35) lvl = 2; else lvl = 3; if (M >= 7.0 && depthKm < 100) lvl = Math.max(lvl, 1); if (M >= 8.0 && depthKm < 60) lvl = Math.max(lvl, 2); if (M >= 8.6 && depthKm < 40) lvl = Math.max(lvl, 3); return lvl; } function tsunamiBaseHeight(M, depthKm){ // meters (very rough, educational) const h0 = 0.08 * Math.pow(10, (M - 6.0)/1.5); const depthF = Math.exp(-depthKm/80); return clamp(h0 * depthF, 0, 30); } function pickAreaGroup(lat, lon){ // choose closer side by representative points let bestPac = 1e9, bestSea = 1e9; for (const a of TSU_AREAS_PAC){ const d = horizKm(lat, lon, a.rep[0], a.rep[1]); if (d < bestPac) bestPac = d; } for (const a of TSU_AREAS_SEA){ const d = horizKm(lat, lon, a.rep[0], a.rep[1]); if (d < bestSea) bestSea = d; } // Okinawa special if (lat < 29.0) return TSU_AREAS_PAC; return (bestPac <= bestSea ? TSU_AREAS_PAC : TSU_AREAS_SEA); } function computeTsunamiState(){ // combine across events: take max per area const on = state.layers.tsunami; if (!on || state.epicenters.length === 0){ state.tsunami.active = false; state.tsunami.level = 0; state.tsunami.label = "なし"; state.tsunami.maxH = 0; state.tsunami.areas = []; return; } const areaMap = new Map(); let globalLevel = 0; let maxH = 0; for (const ev of state.epicenters){ const p = (state.settings.viewMode === "truth") ? globalLevel : 0; state.tsunami.label = TSU_LEVELS[state.tsunami.level].label; state.tsunami.maxH = state.tsunami.active ? maxH : 0; state.tsunami.areas = areas; state.tsunami.updatedAt = performance.now(); } function renderTsunamiPanel(){ if (!dom.tsunamiCard) return; if (!state.layers.tsunami){ dom.tsunamiBadge.textContent = "表示OFF"; dom.tsunamiSummary.textContent = "—"; dom.tsunamiList.innerHTML = ""; return; } if (!state.tsunami.active){ dom.tsunamiBadge.textContent = "なし"; dom.tsunamiBadge.style.background = "rgba(255,255,255,.06)"; dom.tsunamiBadge.style.borderColor = "rgba(255,255,255,.18)"; dom.tsunamiBadge.style.color = "rgba(255,255,255,.85)"; dom.tsunamiSummary.textContent = "津波なし（陸地または条件未満）"; dom.tsunamiList.innerHTML = ""; return; } const col = tsunamiColor(state.tsunami.level); dom.tsunamiBadge.textContent = TSU_LEVELS[state.tsunami.level].short; dom.tsunamiBadge.style.background = `linear-gradient(90deg, ${withAlpha(col,0.55)}, ${withAlpha(col,0.18)} 70%), rgba(255,255,255,.04)`; dom.tsunamiBadge.style.borderColor = withAlpha(col, 0.55); dom.tsunamiBadge.style.color = "rgba(255,255,255,.95)"; dom.tsunamiSummary.textContent = `${state.tsunami.label}（最大 ${fmtM(state.tsunami.maxH)}m）`; const rows = state.tsunami.areas.slice(0, 8).map(a=>{ return `
